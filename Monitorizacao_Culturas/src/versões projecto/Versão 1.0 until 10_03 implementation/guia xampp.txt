--------------------Criar tabela-----------------------
create table Investigador(

IDInvestigador int not null AUTO_INCREMENT, 

email varchar(50) not null,

Nomeinvestigador varchar(100),

CategoriaProfe varchar(100), 

PRIMARY KEY(IDInvestigador) 

);

---------------Criação de um Storage Proceder------------


mysql> delimiter +
create procedure Listar_Investigador()
begin
select NomeInvestigador as nome, Email,CategoriaProfe as Categoria
from Investigador;
end+
delimiter ;

-----------------Usar Storage Procedure----------------------

call Listar_Investigador();

----------------Mostrar Storage Procedure------------------
show procedure status;

----------------Apagar Storage Proceder---------------------

DROP PROCEDURE Listar_Investigador;
