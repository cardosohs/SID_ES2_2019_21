CREATE TRIGGER `medicoesluz_AFTER_INSERT` AFTER INSERT ON `medicoesluz`
 FOR EACH ROW BEGIN
	DECLARE LimiteInf Double;
	DECLARE LimiteSup Double;
-- obtem os valores da tabela sistema
set limiteinf = (select limiteinferiorLuz from sistema limit 1);
set limitesup = (select limiteSuperiorLuz from sistema limit 1);
-- verifica se o novo valor ultrapassa algum

	if (new.valorMedicaoLuz > limitesup ) then 
		  insert into alertas (DataHora,NomeVariavel, LimiteInferior, LimiteSuperior, ValorMedicao,Descricao)
			values (new.DataHoraMed,"Luminosidade",LimiteInf,LimiteSup, new.ValorMedicaoLuz,"Acima do limite");
	end if;
    if (new.valorMedicaoLuz < limiteinf ) then 
      insert into alertas (DataHora,NomeVariavel, LimiteInferior, LimiteSuperior, ValorMedicao,Descricao)
		values (new.DataHoraMed,"Luminosidade",LimiteInf,LimiteSup, new.ValorMedicaoLuz,"Abaixo do limite");
end if;

END

CREATE TRIGGER `processa` AFTER INSERT ON `sensormigracaolog`
 FOR EACH ROW BEGIN
CASE
	WHEN new.sensor="cell" THEN CALL sp_processaLuz(new.momento,new.ordem, new.sensor, new.medicao, new.datetime_sensor);
   WHEN new.sensor='tmp' then CALL sp_processaTemp(new.momento,new.ordem, new.sensor, new.medicao, new.datetime_sensor);
end case;
END
